cmake_minimum_required(VERSION 3.16)
project(cv9x_module32 LANGUAGES CXX)
enable_testing()
add_executable(cv9x_module32_zero_shot zero_shot.cpp)
target_compile_features(cv9x_module32_zero_shot PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(cv9x_module32_zero_shot PRIVATE /W4)
else()
  target_compile_options(cv9x_module32_zero_shot PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME module32_synthetic COMMAND cv9x_module32_zero_shot)

# Build genuine ONNX Runtime inference only when a native SDK is configured.
# Root: cmake -S . -B build-native -DCV9X_ONNXRUNTIME_ROOT=/path/to/onnxruntime
set(CV9X_ONNXRUNTIME_ROOT "" CACHE PATH "Path to native ONNX Runtime SDK")
if(CV9X_ONNXRUNTIME_ROOT)
  if(NOT EXISTS "${CV9X_ONNXRUNTIME_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "ONNX Runtime C++ SDK headers not found")
  endif()
  find_library(CV9X_ONNXRUNTIME_LIB NAMES onnxruntime
    PATHS "${CV9X_ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  add_executable(cv9x_module32_onnx onnx_infer.cpp)
  target_compile_features(cv9x_module32_onnx PRIVATE cxx_std_17)
  target_include_directories(cv9x_module32_onnx PRIVATE "${CV9X_ONNXRUNTIME_ROOT}/include")
  target_link_libraries(cv9x_module32_onnx PRIVATE "${CV9X_ONNXRUNTIME_LIB}")
  set_target_properties(cv9x_module32_onnx PROPERTIES BUILD_RPATH "${CV9X_ONNXRUNTIME_ROOT}/lib")
  if(MSVC)
    target_compile_options(cv9x_module32_onnx PRIVATE /W4)
  else()
    target_compile_options(cv9x_module32_onnx PRIVATE -Wall -Wextra -Wpedantic)
    if(CV9X_WARNINGS_AS_ERRORS)
      target_compile_options(cv9x_module32_onnx PRIVATE -Werror)
    endif()
  endif()
endif()
